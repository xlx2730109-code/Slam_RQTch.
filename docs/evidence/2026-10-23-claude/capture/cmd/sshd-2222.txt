$ systemctl cat sshd-2222.service --no-pager
# /etc/systemd/system/sshd-2222.service
[Unit]
Description=Secondary sshd on port 2222
After=network.target

[Service]
Type=simple
ExecStart=/usr/sbin/sshd -D -p 2222 -o PidFile=/run/sshd-2222.pid
Restart=always
RestartSec=2

[Install]
WantedBy=multi-user.target
[exit=0]
